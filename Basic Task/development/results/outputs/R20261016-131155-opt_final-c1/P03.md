## ✅ 函数实现

```python
def aggregate_by_owner(records):
    """
    按处理人统计记录总数，以及 open、processing、closed 三种状态的数量。
    状态匹配忽略大小写；缺失或为空的 owner 归为 unassigned；
    未知状态计入 other；输入为空时返回空字典。
    """
    if not records:
        return {}

    result = {}

    for record in records:
        owner = record.get('owner')
        status = record.get('status', '').lower()

        # 处理 owner
        if owner is None or owner == '':
            owner = 'unassigned'
        else:
            owner = owner.strip()

        # 初始化该 owner 的统计
        if owner not in result:
            result[owner] = {
                'total': 0,
                'open': 0,
                'processing': 0,
                'closed': 0,
                'other': 0
            }

        # 更新总数
        result[owner]['total'] += 1

        # 根据状态更新对应计数
        if status == 'open':
            result[owner]['open'] += 1
        elif status == 'processing':
            result[owner]['processing'] += 1
        elif status == 'closed':
            result[owner]['closed'] += 1
        else:
            result[owner]['other'] += 1

    return result
```
